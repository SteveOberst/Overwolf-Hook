cmake_minimum_required(VERSION 3.25)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_CURRENT_BINARY_DIR)
    message(FATAL_ERROR "Use a separate build directory, for example cmake -S . -B build/local")
endif()

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/version.txt")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/version.txt" OWH_VERSION LIMIT_COUNT 1)
string(REGEX MATCH "^[0-9]+[.][0-9]+[.][0-9]+" OWH_VERSION_BASE "${OWH_VERSION}")
project(overwolf_hook VERSION ${OWH_VERSION_BASE} LANGUAGES C CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Overwolf-Hook supports Windows x64 only")
endif()

option(BUILD_TESTING "Build the owned Overwolf-Hook test fixtures" ${PROJECT_IS_TOP_LEVEL})
option(OWH_BUILD_TOOLS "Build internal maintainer diagnostic and recording tools" OFF)
option(OWH_HOOK_TESTS "Test MinHook against an owned fixture" ON)
set(OWH_REVIEWED_CONSUMER "" CACHE FILEPATH "Optional consumer DLL loaded only by the owned tests")

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(OverwolfHookTargetOptions)
include(OverwolfHookBuildInfo)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

include(MinHook)
include(BoostInterprocess)
set(OWH_NATIVE_DIRECTORY "${PROJECT_BINARY_DIR}/native/$<CONFIG>")
add_subdirectory(modules)
if(OWH_BUILD_TOOLS)
    add_subdirectory(tools)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()

include(OverwolfHookPackaging)
