cmake_minimum_required(VERSION 3.25)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/version.txt" OEH_VERSION LIMIT_COUNT 1)
set(OWH_RELEASE_VERSION "${OEH_VERSION}")
string(REGEX MATCH "^[0-9]+[.][0-9]+[.][0-9]+" OWH_PROJECT_VERSION "${OWH_RELEASE_VERSION}")
project(overwolf_event_hook VERSION ${OWH_PROJECT_VERSION} LANGUAGES C CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Overwolf-Event-Hook supports Windows x64 only")
endif()

include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(OEH_HOOK_TESTS "Build MinHook experiments against the owned fixture" ON)
option(OEH_FFM_TESTS "Exercise the C ABI from Java 25 FFM" OFF)
option(OEH_OVERWOLF_PROBE "Build the reviewed, explicit Overwolf capture experiment" ON)
set(OEH_REVIEWED_CONSUMER "" CACHE FILEPATH "Optional exact consumer DLL for the owned-host ABI test")
if(OEH_OVERWOLF_PROBE OR (BUILD_TESTING AND OEH_HOOK_TESTS))
    include(cmake/MinHook.cmake)
endif()

add_library(oeh_transport STATIC src/transport/channel.cpp)
target_compile_features(oeh_transport PUBLIC cxx_std_20)
target_include_directories(oeh_transport PUBLIC include PRIVATE src)

add_library(oeh_capture SHARED src/capture/capture.cpp)
add_library(OverwolfEventHook::Capture ALIAS oeh_capture)
add_library(oeh_client SHARED src/client/client.cpp)
add_library(OverwolfEventHook::Client ALIAS oeh_client)
foreach(target IN ITEMS oeh_capture oeh_client)
    target_link_libraries(${target} PRIVATE oeh_transport)
    target_include_directories(${target} PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src)
    target_compile_definitions(${target} PRIVATE OEH_BUILD_${target})
    set_target_properties(${target} PROPERTIES PREFIX "")
endforeach()
set_target_properties(oeh_capture PROPERTIES EXPORT_NAME Capture)
set_target_properties(oeh_client PROPERTIES EXPORT_NAME Client)
if(OEH_OVERWOLF_PROBE)
    target_sources(oeh_capture PRIVATE src/capture/overwolf/probe.cpp)
    target_link_libraries(oeh_capture PRIVATE minhook bcrypt)
    add_executable(oeh_probe tools/probe.cpp)
    target_compile_features(oeh_probe PRIVATE cxx_std_20)
    target_compile_definitions(oeh_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(oeh_probe PRIVATE oeh_client)
    add_dependencies(oeh_probe oeh_capture)
    install(TARGETS oeh_probe RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES third_party/MinHook.LICENSE.txt
        DESTINATION ${CMAKE_INSTALL_DOCDIR}/third-party RENAME MinHook-LICENSE.txt)
endif()

foreach(target IN ITEMS oeh_transport oeh_capture oeh_client)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
    endif()
endforeach()

if(BUILD_TESTING)
    add_executable(oeh_transport_tests tests/transport_tests.cpp)
    target_compile_features(oeh_transport_tests PRIVATE cxx_std_20)
    target_link_libraries(oeh_transport_tests PRIVATE oeh_capture oeh_client)
    target_compile_definitions(oeh_transport_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    foreach(case IN ITEMS roundtrip readers boundary overflow buffer invalid lifecycle process crash corrupt)
        add_test(NAME transport.${case} COMMAND oeh_transport_tests ${case})
        set_tests_properties(transport.${case} PROPERTIES TIMEOUT 15)
    endforeach()

    if(OEH_FFM_TESTS)
        find_package(Java 25 REQUIRED COMPONENTS Runtime)
        add_test(NAME java_ffm COMMAND ${Java_JAVA_EXECUTABLE}
            --enable-native-access=ALL-UNNAMED ${CMAKE_CURRENT_SOURCE_DIR}/examples/java/FfmRoundtrip.java
            $<TARGET_FILE:oeh_capture> $<TARGET_FILE:oeh_client>)
        set_tests_properties(java_ffm PROPERTIES TIMEOUT 30)
    endif()

    if(OEH_HOOK_TESTS)
        add_library(oeh_hook_fixture SHARED tests/hook_fixture.cpp)
        if(MSVC)
            target_compile_options(oeh_hook_fixture PRIVATE /Od /Ob0 /GL-)
        endif()
        add_executable(oeh_hook_tests tests/hook_tests.cpp)
        target_compile_features(oeh_hook_tests PRIVATE cxx_std_20)
        target_compile_definitions(oeh_hook_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_link_libraries(oeh_hook_tests PRIVATE minhook oeh_capture oeh_client)
        add_dependencies(oeh_hook_tests oeh_hook_fixture)
        add_test(NAME hook_trampoline COMMAND oeh_hook_tests $<TARGET_FILE:oeh_hook_fixture>)
        set_tests_properties(hook_trampoline PROPERTIES TIMEOUT 15)
    endif()
    if(OEH_OVERWOLF_PROBE AND OEH_REVIEWED_CONSUMER)
        add_executable(oeh_consumer_host tests/consumer_host.cpp)
        target_compile_features(oeh_consumer_host PRIVATE cxx_std_20)
        target_compile_definitions(oeh_consumer_host PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_include_directories(oeh_consumer_host PRIVATE src)
        target_link_libraries(oeh_consumer_host PRIVATE oeh_capture oeh_client)
        add_test(NAME reviewed_consumer_hook COMMAND oeh_consumer_host "${OEH_REVIEWED_CONSUMER}")
        set_tests_properties(reviewed_consumer_hook PROPERTIES TIMEOUT 20)
    endif()
    if(OEH_OVERWOLF_PROBE)
        add_executable(oeh_channel_tests tests/channel_tests.cpp)
        target_compile_features(oeh_channel_tests PRIVATE cxx_std_20)
        target_include_directories(oeh_channel_tests PRIVATE src)
        add_test(NAME event_channels COMMAND oeh_channel_tests)
    endif()
endif()

install(TARGETS oeh_capture oeh_client EXPORT OverwolfEventHookTargets
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/overwolf_event_hook/bridge.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/overwolf_event_hook)
if(OEH_OVERWOLF_PROBE)
    install(FILES include/overwolf_event_hook/probe.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/overwolf_event_hook)
endif()
install(EXPORT OverwolfEventHookTargets NAMESPACE OverwolfEventHook:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OverwolfEventHook)
configure_package_config_file(cmake/OverwolfEventHookConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/OverwolfEventHookConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OverwolfEventHook)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/OverwolfEventHookConfigVersion.cmake" "set(PACKAGE_VERSION \"${OWH_RELEASE_VERSION}\")\nset(PACKAGE_VERSION_COMPATIBLE FALSE)\nset(PACKAGE_VERSION_EXACT FALSE)\nif(DEFINED CMAKE_SIZEOF_VOID_P AND NOT CMAKE_SIZEOF_VOID_P STREQUAL \"${CMAKE_SIZEOF_VOID_P}\")\n set(PACKAGE_VERSION_UNSUITABLE TRUE)\nendif()\n")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/OverwolfEventHookConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/OverwolfEventHookConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OverwolfEventHook)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "overwolf-event-hook-${OWH_RELEASE_VERSION}-windows-x64")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGE_VENDOR "SteveOberst")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/SteveOberst/Overwolf-Event-Hook")
include(CPack)
