cmake_minimum_required(VERSION 3.25)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/version.txt" OET_VERSION LIMIT_COUNT 1)
project(overwolf_hook VERSION ${OET_VERSION} LANGUAGES C CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Overwolf-Hook supports Windows x64 only")
endif()

include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(OET_HOOK_TESTS "Build MinHook experiments against the owned fixture" ON)
option(OET_FFM_TESTS "Exercise the C ABI from Java 25 FFM" OFF)
option(OET_OVERWOLF_PROBE "Build the reviewed, explicit Overwolf capture experiment" ON)
set(OET_REVIEWED_CONSUMER "" CACHE FILEPATH "Optional exact consumer DLL for the owned-host ABI test")
if(OET_OVERWOLF_PROBE OR (BUILD_TESTING AND OET_HOOK_TESTS))
    include(cmake/MinHook.cmake)
endif()

add_library(oet_transport STATIC src/transport/channel.cpp)
target_compile_features(oet_transport PUBLIC cxx_std_20)
target_include_directories(oet_transport PUBLIC include PRIVATE src)

add_library(oet_capture SHARED src/capture/capture.cpp)
add_library(EventTap::Capture ALIAS oet_capture)
add_library(oet_client SHARED src/client/client.cpp)
add_library(EventTap::Client ALIAS oet_client)
foreach(target IN ITEMS oet_capture oet_client)
    target_link_libraries(${target} PRIVATE oet_transport)
    target_include_directories(${target} PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src)
    target_compile_definitions(${target} PRIVATE OET_BUILD_${target})
    set_target_properties(${target} PROPERTIES PREFIX "")
endforeach()
set_target_properties(oet_capture PROPERTIES EXPORT_NAME Capture)
set_target_properties(oet_client PROPERTIES EXPORT_NAME Client)
if(OET_OVERWOLF_PROBE)
    target_sources(oet_capture PRIVATE src/capture/overwolf/probe.cpp)
    target_link_libraries(oet_capture PRIVATE minhook bcrypt)
    add_executable(oet_probe tools/probe.cpp)
    target_compile_features(oet_probe PRIVATE cxx_std_20)
    target_compile_definitions(oet_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(oet_probe PRIVATE oet_client)
    add_dependencies(oet_probe oet_capture)
    install(TARGETS oet_probe RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES third_party/MinHook.LICENSE.txt
        DESTINATION ${CMAKE_INSTALL_DOCDIR}/third-party RENAME MinHook-LICENSE.txt)
endif()

foreach(target IN ITEMS oet_transport oet_capture oet_client)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
    endif()
endforeach()

if(BUILD_TESTING)
    add_executable(oet_transport_tests tests/transport_tests.cpp)
    target_compile_features(oet_transport_tests PRIVATE cxx_std_20)
    target_link_libraries(oet_transport_tests PRIVATE oet_capture oet_client)
    target_compile_definitions(oet_transport_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    foreach(case IN ITEMS roundtrip readers boundary overflow buffer invalid lifecycle process crash corrupt)
        add_test(NAME transport.${case} COMMAND oet_transport_tests ${case})
        set_tests_properties(transport.${case} PROPERTIES TIMEOUT 15)
    endforeach()

    if(OET_FFM_TESTS)
        find_package(Java 25 REQUIRED COMPONENTS Runtime)
        add_test(NAME java_ffm COMMAND ${Java_JAVA_EXECUTABLE}
            --enable-native-access=ALL-UNNAMED ${CMAKE_CURRENT_SOURCE_DIR}/examples/java/FfmRoundtrip.java
            $<TARGET_FILE:oet_capture> $<TARGET_FILE:oet_client>)
        set_tests_properties(java_ffm PROPERTIES TIMEOUT 30)
    endif()

    if(OET_HOOK_TESTS)
        add_library(oet_hook_fixture SHARED tests/hook_fixture.cpp)
        if(MSVC)
            target_compile_options(oet_hook_fixture PRIVATE /Od /Ob0 /GL-)
        endif()
        add_executable(oet_hook_tests tests/hook_tests.cpp)
        target_compile_features(oet_hook_tests PRIVATE cxx_std_20)
        target_compile_definitions(oet_hook_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_link_libraries(oet_hook_tests PRIVATE minhook oet_capture oet_client)
        add_dependencies(oet_hook_tests oet_hook_fixture)
        add_test(NAME hook_trampoline COMMAND oet_hook_tests $<TARGET_FILE:oet_hook_fixture>)
        set_tests_properties(hook_trampoline PROPERTIES TIMEOUT 15)
    endif()
    if(OET_OVERWOLF_PROBE AND OET_REVIEWED_CONSUMER)
        add_executable(oet_consumer_host tests/consumer_host.cpp)
        target_compile_features(oet_consumer_host PRIVATE cxx_std_20)
        target_compile_definitions(oet_consumer_host PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_include_directories(oet_consumer_host PRIVATE src)
        target_link_libraries(oet_consumer_host PRIVATE oet_capture oet_client)
        add_test(NAME reviewed_consumer_hook COMMAND oet_consumer_host "${OET_REVIEWED_CONSUMER}")
        set_tests_properties(reviewed_consumer_hook PROPERTIES TIMEOUT 20)
    endif()
    if(OET_OVERWOLF_PROBE)
        add_executable(oet_channel_tests tests/channel_tests.cpp)
        target_compile_features(oet_channel_tests PRIVATE cxx_std_20)
        target_include_directories(oet_channel_tests PRIVATE src)
        add_test(NAME event_channels COMMAND oet_channel_tests)
    endif()
endif()

install(TARGETS oet_capture oet_client EXPORT EventTapTargets
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/event_tap/bridge.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/event_tap)
if(OET_OVERWOLF_PROBE)
    install(FILES include/event_tap/probe.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/event_tap)
endif()
install(EXPORT EventTapTargets NAMESPACE EventTap:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EventTap)
configure_package_config_file(cmake/EventTapConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/EventTapConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EventTap)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/EventTapConfigVersion.cmake
    COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/EventTapConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/EventTapConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EventTap)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "overwolf-hook-${PROJECT_VERSION}-windows-x64")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGE_VENDOR "SteveOberst")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/SteveOberst/Overwolf-Hook")
include(CPack)
